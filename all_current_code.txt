/// Renders an SVG string from AppSvgs in a single color.
class AppIcon extends StatelessWidget {
  const AppIcon(this.svg, {super.key, this.size = 20, this.color});

  final String svg;
  final double size;
  final Color? color;

  @override
  Widget build(BuildContext context) {
    final c = color ?? IconTheme.of(context).color ?? AppColors.text;
    return SvgPicture.string(
      svg,
      width: size,
      height: size,
      colorFilter: ColorFilter.mode(c, BlendMode.srcIn),
    );
  }
}

/// White rounded card with a thin border.
class AppCard extends StatelessWidget {
  const AppCard({
    super.key,
    required this.child,
    this.padding = const EdgeInsets.all(20),
    this.onTap,
  });

  final Widget child;
  final EdgeInsetsGeometry padding;
  final VoidCallback? onTap;

  @override
  Widget build(BuildContext context) {
    return Material(
      color: Colors.white,
      clipBehavior: Clip.antiAlias,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(20),
        side: const BorderSide(color: AppColors.border),
      ),
      child: InkWell(
        onTap: onTap,
        child: Padding(padding: padding, child: child),
      ),
    );
  }
}

/// Centers page content and limits its width on big screens.

  const PageContainer({super.key, required this.child, this.maxWidth = 1200});

  final Widget child;
  final double maxWidth;

  @override

    final mobile = Responsive.isMobile(context);
    return Align(

/// Item image from a URL. Shows a placeholder when empty or broken.
/// Use expand: true inside an AspectRatio to fill the space.
class ItemThumb extends StatelessWidget {
  const ItemThumb({super.key, this.url, this.size = 48, this.expand = false});

  final String? url;
  final double size;
  final bool expand;

  @override
  Widget build(BuildContext context) {
    final placeholder = Stack(
      alignment: Alignment.center,
      children: [
        Center(
          child: AppIcon(AppSvgs.imageIcon, size: size * 0.4, color: AppColors.textSoft),
        ),
        // if (url != null)
        //   Padding(
        //     padding: EdgeInsetsGeometry.all(10),
        //     child: Text(
        //       url ?? 'N/A',
        //     ),
        //   ),
      ],
    );
    final hasUrl = url != null && url!.trim().isNotEmpty;

    return ClipRRect(
      borderRadius: BorderRadius.circular(expand ? 14 : size * 0.25),
      child: Container(
        width: expand ? null : size,
        height: expand ? null : size,
        color: AppColors.bg,
        child: hasUrl
            ? Image.network(
                url!,
                fit: BoxFit.contain,
                errorBuilder: (e1, e2, e3) {
                  return placeholder;
                },
              )
            : placeholder,
      ),
    );
  }
}

/// 3 -> "+3", -4 -> "-4"
String signed(int n) => n > 0 ? '+$n' : '$n';

void showMessage(BuildContext context, String text, {bool isError = false}) {
  ScaffoldMessenger.of(context)
    ..hideCurrentSnackBar()
    ..showSnackBar(
      SnackBar(
        content: Text(text),
        backgroundColor: isError ? AppColors.danger : AppColors.text,
        behavior: SnackBarBehavior.floating,
      ),
    );
}

/// Go back if there is a page to go back to, otherwise open [fallback].
void goBackOr(BuildContext context, String fallback) {
  if (context.canPop()) {
    context.pop();
  } else {
    context.go(fallback);
  }
}

class AppTheme {
  static ThemeData get light {
    final scheme =
        ColorScheme.fromSeed(
          seedColor: AppColors.primary,
          brightness: Brightness.light,
        ).copyWith(
          primary: AppColors.primary,
          onPrimary: Colors.white,
          surface: AppColors.surface,
          onSurface: AppColors.text,
          error: AppColors.danger,
        );

    final textTheme = GoogleFonts.nunitoSansTextTheme(
      ThemeData(brightness: Brightness.light).textTheme,
    ).apply(bodyColor: AppColors.text, displayColor: AppColors.text);

    OutlineInputBorder border(Color color) => OutlineInputBorder(
      borderRadius: BorderRadius.circular(14),
      borderSide: BorderSide(color: color),
    );

    return ThemeData(
      useMaterial3: true,
      colorScheme: scheme,
      textTheme: textTheme,
      scaffoldBackgroundColor: AppColors.bg,
      dividerColor: AppColors.border,
      appBarTheme: const AppBarTheme(
        backgroundColor: AppColors.bg,
        foregroundColor: AppColors.text,
        elevation: 0,
        scrolledUnderElevation: 0,
        centerTitle: false,
      ),
      inputDecorationTheme: InputDecorationTheme(
        filled: true,
        fillColor: Colors.white,
        contentPadding: const EdgeInsets.symmetric(horizontal: 16, vertical: 14),
        border: border(AppColors.border),
        enabledBorder: border(AppColors.border),
        focusedBorder: border(AppColors.primary),
        errorBorder: border(AppColors.danger),
        focusedErrorBorder: border(AppColors.danger),
      ),
      filledButtonTheme: FilledButtonThemeData(
        style: FilledButton.styleFrom(
          backgroundColor: AppColors.primary,
          foregroundColor: Colors.white,
          minimumSize: const Size(48, 48),
          shape: const StadiumBorder(),
        ),
      ),
      outlinedButtonTheme: OutlinedButtonThemeData(
        style: OutlinedButton.styleFrom(
          foregroundColor: AppColors.primary,
          side: const BorderSide(color: AppColors.primary),
          minimumSize: const Size(48, 48),
          shape: const StadiumBorder(),
        ),
      ),
      textButtonTheme: TextButtonThemeData(
        style: TextButton.styleFrom(foregroundColor: AppColors.primary),
      ),
      navigationBarTheme: const NavigationBarThemeData(
        backgroundColor: Colors.white,
        indicatorColor: AppColors.primarySoft,
      ),
    );
  }
}

enum TransitionType {
  none,
  fade,
  slide,
  scale,
  fadeSlide,
}

CustomTransitionPage<void> _buildTransitionPage({
  required GoRouterState state,
  required Widget child,
  TransitionType transition = TransitionType.none,
}) {
  return CustomTransitionPage<void>(
    key: state.pageKey,
    child: child,
    transitionDuration: const Duration(milliseconds: 250),
    reverseTransitionDuration: const Duration(milliseconds: 200),
    transitionsBuilder: (context, animation, secondaryAnimation, child) {
      switch (transition) {
        case TransitionType.none:
          return child;

        case TransitionType.fade:
          return FadeTransition(
            opacity: animation,
            child: child,
          );

        case TransitionType.slide:
          return SlideTransition(
            position: animation.drive(
              Tween<Offset>(
                begin: const Offset(1, 0),
                end: Offset.zero,
              ).chain(
                CurveTween(curve: Curves.easeInOut),
              ),
            ),
            child: child,
          );

        case TransitionType.scale:
          return ScaleTransition(
            scale: animation.drive(
              Tween<double>(
                begin: 0.9,
                end: 1,
              ).chain(
                CurveTween(curve: Curves.easeOut),
              ),
            ),
            child: child,
          );

        case TransitionType.fadeSlide:
          final slideAnimation = animation.drive(
            Tween<Offset>(
              begin: const Offset(0.08, 0),
              end: Offset.zero,
            ).chain(
              CurveTween(curve: Curves.easeOut),
            ),
          );

          return FadeTransition(
            opacity: animation,
            child: SlideTransition(
              position: slideAnimation,
              child: child,
            ),
          );
      }
    },
  );
}

final routerProvider = Provider<GoRouter>((ref) {
  // Refresh redirects when authentication state changes.
  final refresh = ValueNotifier<int>(0);

  ref.listen<AppUser?>(
    authProvider,
    (_, __) => refresh.value++,
  );

  ref.onDispose(refresh.dispose);

  return GoRouter(
    initialLocation: '/',
    refreshListenable: refresh,
    redirect: (context, state) {
      final loggedIn = ref.read(authProvider) != null;
      final atLogin = state.matchedLocation == '/login';

      if (!loggedIn) {
        return atLogin ? null : '/login';
      }

      if (atLogin) {
        return '/';
      }

      return null;
    },
    routes: [
      // Authentication
      GoRoute(
        path: '/login',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: const LoginPage(),
          transition: TransitionType.fade,
        ),
      ),

      // Home. No sidebar or bottom bar. Everything opens from the action tiles.
      GoRoute(
        path: '/',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: const DashboardPage(),
          transition: TransitionType.fade,
        ),
      ),

      // Screens opened from the dashboard actions (with a back arrow)
      GoRoute(
        path: '/items',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: const SubPage(child: ItemsPage()),
          transition: TransitionType.slide,
        ),
      ),
      GoRoute(
        path: '/history',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: const SubPage(child: HistoryPage()),
          transition: TransitionType.slide,
        ),
      ),
      GoRoute(
        path: '/reports',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: const SubPage(child: ReportsPage()),
          transition: TransitionType.slide,
        ),
      ),

      // Item screens
      GoRoute(
        path: '/items/new',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: const ItemFormPage(),
          transition: TransitionType.slide,
        ),
      ),
      GoRoute(
        path: '/items/:id',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: ItemDetailPage(
            itemId: state.pathParameters['id']!,
          ),
          transition: TransitionType.slide,
        ),
      ),
      GoRoute(
        path: '/items/:id/edit',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: ItemFormPage(
            itemId: state.pathParameters['id'],
          ),
          transition: TransitionType.slide,
        ),
      ),

      // Supply, sale and manual orders
      GoRoute(
        path: '/orders/:type',
        pageBuilder: (context, state) => _buildTransitionPage(
          state: state,
          child: OrderFormPage(
            type: EntryType.values.asNameMap()[state.pathParameters['type']] ?? EntryType.supply,
            presetItemId: state.uri.queryParameters['item'],
          ),
          transition: TransitionType.slide,
        ),
      ),
    ],
  );
});

import 'package:supabase_flutter/supabase_flutter.dart' show PostgrestException;

class AppException implements Exception {
  const AppException(this.message);
  final String message;

  @override
  String toString() => message;
}

/// Turns any error into a short message that is safe to show to the user.
String errorMessage(Object error) {
  if (error is AppException) return error.message;
  if (error is PostgrestException) {
    if (error.code == '23505') {
      return 'This value already exists. Please use a different one.';
    }
    return error.message;
  }
  return 'Something went wrong. Please try again.';
}

import 'dart:js_interop';
import 'dart:typed_data';

import 'package:web/web.dart' as web;

/// Starts a file download in the browser (Flutter web only).
void downloadBytes(
  Uint8List bytes,
  String fileName, {
  String mimeType = 'application/octet-stream',
}) {
  final blob = web.Blob(
    [bytes.toJS].toJS,
    web.BlobPropertyBag(type: mimeType),
  );
  final url = web.URL.createObjectURL(blob);

  final anchor = web.document.createElement('a') as web.HTMLAnchorElement;
  anchor.href = url;
  anchor.download = fileName;
  anchor.style.display = 'none';

  web.document.body!.appendChild(anchor);
  anchor.click();
  anchor.remove();

  // Give the browser a moment to start the download before freeing the file.
  Future<void>.delayed(const Duration(seconds: 2), () {
    web.URL.revokeObjectURL(url);
  });
}

final itemRepositoryProvider = Provider((ref) => ItemRepository(ref.watch(supabaseProvider)));
final inventoryRepositoryProvider = Provider((ref) => InventoryRepository(ref.watch(supabaseProvider)));

// ---------- Auth ----------

class AuthNotifier extends Notifier<AppUser?> {
  static const _key = 'app_user';

  @override
  AppUser? build() {
    final raw = ref.read(sharedPrefsProvider).getString(_key);
    if (raw == null) return null;
    try {
      return AppUser.fromJson(jsonDecode(raw) as Map<String, dynamic>);
    } catch (_) {
      return null;
    }
  }

  Future<void> login(String username, String password) async {
    final user = await ref.read(authRepositoryProvider).login(username, password);
    await ref.read(sharedPrefsProvider).setString(_key, jsonEncode(user.toJson()));
    state = user;
  }

  Future<void> logout() async {
    await ref.read(sharedPrefsProvider).remove(_key);
    state = null;
  }
}

final authProvider = NotifierProvider<AuthNotifier, AppUser?>(AuthNotifier.new);

typedef HistoryQuery = ({EntryType? type, DateTime? from, DateTime? to});

final historyProvider = FutureProvider.autoDispose.family<List<InventoryEntry>, HistoryQuery>(
  (ref, q) => ref.watch(inventoryRepositoryProvider).fetchEntries(from: q.from, to: q.to, type: q.type),

  return buildReportSheets(from: q.from, to: q.to, serverRows: rows, activeItems: items);

  ref.invalidate(brandsProvider);
  ref.invalidate(categoriesProvider);
  ref.invalidate(entriesByDayProvider);
  ref.invalidate(historyProvider);
  ref.invalidate(itemHistoryProvider);
  ref.invalidate(stockReportProvider);
}

class LoginPage extends ConsumerStatefulWidget {
  const LoginPage({super.key});

  @override
  ConsumerState<LoginPage> createState() => _LoginPageState();
}

class _LoginPageState extends ConsumerState<LoginPage> {
  final _username = TextEditingController();
  final _password = TextEditingController();
  bool _busy = false;
  bool _hidePassword = true;
  String? _error;

  @override
  void dispose() {
    _username.dispose();
    _password.dispose();
    super.dispose();
  }

  Future<void> _submit() async {
    final username = _username.text.trim();
    final password = _password.text;
    if (username.isEmpty || password.isEmpty) {
      setState(() => _error = 'Enter your username and password.');
      return;
    }

    setState(() {
      _busy = true;
      _error = null;
    });
    try {
      await ref.read(authProvider.notifier).login(username, password);
      // On success the router moves to the dashboard.
    } catch (e) {
      if (mounted) setState(() => _error = errorMessage(e));
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(20),
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 400),
            child: AppCard(
              padding: const EdgeInsets.all(28),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                crossAxisAlignment: CrossAxisAlignment.stretch,
                children: [
                  const Text(
                    'Bar Inventory',
                    style: TextStyle(fontSize: 24, fontWeight: FontWeight.w700),
                  ),
                  const SizedBox(height: 4),
                  Text(
                    AppEnv.isProd ? 'Sign in to continue' : 'Sign in to continue (${AppEnv.envName})',
                    style: const TextStyle(color: AppColors.textSoft),
                  ),
                  const SizedBox(height: 24),
                  TextField(
                    controller: _username,
                    enabled: !_busy,
                    autofocus: true,
                    textInputAction: TextInputAction.next,
                    autofillHints: const [AutofillHints.username],
                    decoration: const InputDecoration(labelText: 'Username'),
                  ),
                  const SizedBox(height: 14),
                  TextField(
                    controller: _password,
                    enabled: !_busy,
                    obscureText: _hidePassword,
                    onSubmitted: (_) => _submit(),
                    autofillHints: const [AutofillHints.password],
                    decoration: InputDecoration(
                      labelText: 'Password',
                      suffix: TextButton(
                        onPressed: () => setState(() => _hidePassword = !_hidePassword),
                        child: Text(_hidePassword ? 'Show' : 'Hide'),
                      ),
                    ),
                  ),
                  if (_error != null) ...[
                    const SizedBox(height: 14),
                    Text(
                      _error!,
                      style: const TextStyle(color: AppColors.danger),
                    ),
                  ],
                  const SizedBox(height: 22),
                  FilledButton(
                    onPressed: _busy ? null : _submit,
                    child: _busy
                        ? const SizedBox(
                            width: 20,
                            height: 20,
                            child: CircularProgressIndicator(
                              strokeWidth: 2,
                              color: Colors.white,
                            ),
                          )
                        : const Text('Log in'),
                  ),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }
}

class HistoryPage extends ConsumerStatefulWidget {
  const HistoryPage({super.key});

  @override
  ConsumerState<HistoryPage> createState() => _HistoryPageState();
}

class _HistoryPageState extends ConsumerState<HistoryPage> {
  static const _limit = 200;
  static final _fmt = DateFormat('d MMM yyyy');

  EntryType? _type;
  // Default: last 30 days. Use "Clear dates" to see everything.
  DateTime? _from = _dateOnly(DateTime.now().subtract(const Duration(days: 30)));
  DateTime? _to;
  String? _busyId; // record that is being deleted

  Future<void> _pickRange() async {
    final today = _dateOnly(DateTime.now());
    final picked = await showDateRangePicker(
      context: context,
      firstDate: DateTime(2020),
      lastDate: today,
      initialDateRange: _from == null ? null : DateTimeRange(start: _from!, end: _to ?? today),
    );
    if (picked != null) {
      setState(() {
        _from = _dateOnly(picked.start);
        _to = _dateOnly(picked.end);
      });
    }
  }

  Future<bool?> _confirmDelete(InventoryEntry e) {
    final what = switch (e.type) {
      EntryType.sale => 'sale',
      EntryType.supply => 'supply',
      EntryType.manual => 'manual adjustment',
    };

    return showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        backgroundColor: Colors.white,
        title: Text('Delete this $what record?'),
        content: SizedBox(
          width: 360,
          child: SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const Text('The record will be removed and the stock will be changed back:'),
                const SizedBox(height: 12),
                for (final l in e.lines)
                  Padding(
                    padding: const EdgeInsets.symmetric(vertical: 3),
                    child: Row(
                      children: [
                        Expanded(
                          child: Text(
                            l.itemSubtitle.isEmpty ? l.itemName : '${l.itemName} (${l.itemSubtitle})',
                          ),
                        ),
                        const SizedBox(width: 8),
                        Text(
                          signed(-l.quantityChange),
                          style: TextStyle(
                            fontWeight: FontWeight.w700,
                            color: -l.quantityChange > 0 ? AppColors.success : AppColors.danger,
                          ),
                        ),
                      ],
                    ),
                  ),
              ],
            ),
          ),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Cancel'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(ctx, true),
            style: TextButton.styleFrom(foregroundColor: AppColors.danger),
            child: const Text('Delete'),
          ),
        ],
      ),
    );
  }

  Future<void> _delete(InventoryEntry e) async {
    final ok = await _confirmDelete(e);
    if (ok != true || !mounted) return;

    final user = ref.read(authProvider);
    if (user == null) return;

    setState(() => _busyId = e.id);
    try {
      await ref.read(inventoryRepositoryProvider).deleteEntry(userId: user.id, entryId: e.id);
      refreshAll(ref);
      if (mounted) showMessage(context, 'Record deleted. Stock updated.');
    } catch (err) {
      if (mounted) showMessage(context, errorMessage(err), isError: true);
    } finally {
      if (mounted) setState(() => _busyId = null);
    }
  }

  @override
  Widget build(BuildContext context) {
    final query = (type: _type, from: _from, to: _to);
    final async = ref.watch(historyProvider(query));

    return PageContainer(
      maxWidth: 900,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          const PageHeader(title: 'Inventory History'),
          const SizedBox(height: 16),
          _filters(),
          const SizedBox(height: 12),
          Expanded(
            child: async.when(
              loading: () => const LoadingView(),
              error: (e, _) => ErrorState(
                error: e,
                onRetry: () => ref.invalidate(historyProvider(query)),
              ),
              data: (entries) {
                if (entries.isEmpty) {
                  return const EmptyState(message: 'No entries found.');
                }
                // Insert a date heading whenever the date changes
                final rows = <Object>[];
                DateTime? last;
                for (final e in entries) {
                  if (last == null || e.entryDate != last) {
                    rows.add(e.entryDate);
                    last = e.entryDate;
                  }
                  rows.add(e);
                }
                final showLimitNote = entries.length >= _limit;

                return ListView.builder(
                  padding: const EdgeInsets.only(bottom: 24),
                  itemCount: rows.length + (showLimitNote ? 1 : 0),
                  itemBuilder: (context, i) {
                    if (i == rows.length) {
                      return const Padding(
                        padding: EdgeInsets.all(12),
                        child: Text(
                          'Showing the latest 200 entries. Narrow the dates to see older ones.',
                          textAlign: TextAlign.center,
                          style: TextStyle(color: AppColors.textSoft),
                        ),
                      );
                    }
                    final row = rows[i];
                    if (row is DateTime) {
                      return Padding(
                        padding: const EdgeInsets.only(top: 12, bottom: 8),
                        child: Text(
                          DateFormat('EEEE, d MMM yyyy').format(row),
                          style: const TextStyle(
                            fontWeight: FontWeight.w600,
                            color: AppColors.textSoft,
                          ),
                        ),
                      );
                    }
                    final entry = row as InventoryEntry;
                    return Padding(
                      padding: const EdgeInsets.only(bottom: 10),
                      child: EntryCard(
                        entry: entry,
                        deleting: _busyId == entry.id,
                        onDelete: _busyId == null ? () => _delete(entry) : null,
                      ),
                    );
                  },
                );
              },
            ),
          ),
        ],
      ),
    );
  }

  Widget _filters() {
    final dateLabel = _from == null ? 'All dates' : '${_fmt.format(_from!)} - ${_to == null ? 'today' : _fmt.format(_to!)}';

    return SingleChildScrollView(
      scrollDirection: Axis.horizontal,
      child: Row(
        children: [
          for (final t in [null, ...EntryType.values])
            Padding(
              padding: const EdgeInsets.only(right: 8),
              child: ChoiceChip(
                label: Text(t?.label ?? 'All'),
                selected: _type == t,
                showCheckmark: false,
                selectedColor: AppColors.primarySoft,
                backgroundColor: Colors.white,
                side: BorderSide(
                  color: _type == t ? AppColors.primary : AppColors.border,
                ),
                onSelected: (_) => setState(() => _type = t),
              ),
            ),
          const SizedBox(width: 8),
          ActionChip(
            avatar: AppIcon(
              AppSvgs.calendarIcon,
              size: 16,
              color: AppColors.textSoft,
            ),
            label: Text(dateLabel),
            backgroundColor: Colors.white,
            side: const BorderSide(color: AppColors.border),
            onPressed: _pickRange,
          ),
          if (_from != null) ...[
            const SizedBox(width: 8),
            ActionChip(
              label: const Text('Clear dates'),
              backgroundColor: Colors.white,
              side: const BorderSide(color: AppColors.border),
              onPressed: () => setState(() {
                _from = null;
                _to = null;
              }),
            ),
          ],
        ],
      ),
    );
  }
}

/// One row in the order: an item and the quantity typed for it.
class _Line {
  _Line(this.item, String initial) : controller = TextEditingController(text: initial);

  final Item item;
  final TextEditingController controller;

  int get qty => int.tryParse(controller.text.trim()) ?? 0;
}

class OrderFormPage extends ConsumerStatefulWidget {
  const OrderFormPage({super.key, required this.type, this.presetItemId});

  final EntryType type;

  /// When opened from Item Detail, this item is added automatically.
  final String? presetItemId;

  @override
  ConsumerState<OrderFormPage> createState() => _OrderFormPageState();
}

class _OrderFormPageState extends ConsumerState<OrderFormPage> {
  final _formKey = GlobalKey<FormState>();
  final _reference = TextEditingController();
  final _note = TextEditingController();
  final Map<String, _Line> _lines = {};
  final List<TextEditingController> _removed = [];
  DateTime _date = _today();
  bool _busy = false;

  static DateTime _today() {
    final n = DateTime.now();
    return DateTime(n.year, n.month, n.day);
  }

  bool get _isManual => widget.type == EntryType.manual;
  bool get _isSale => widget.type == EntryType.sale;

  String get _title => switch (widget.type) {
    EntryType.supply => 'New supply order',
    EntryType.sale => 'New sale order',
    EntryType.manual => 'Adjust stock',
  };

  String get _hint => switch (widget.type) {
    EntryType.supply => 'Items you received. Stock goes up.',
    EntryType.sale => 'Items sold. Stock goes down.',
    EntryType.manual => 'Enter how much to add (+) or remove (-) for each item.',
  };

  String get _saveLabel => switch (widget.type) {
    EntryType.supply => 'Save supply order',
    EntryType.sale => 'Save sale order',
    EntryType.manual => 'Save adjustment',
  };

  @override
  void initState() {
    super.initState();
    final id = widget.presetItemId;
    if (id != null && id.isNotEmpty) _loadPreset(id);
  }

  @override
  void dispose() {
    _reference.dispose();
    _note.dispose();
    for (final l in _lines.values) {
      l.controller.dispose();
    }
    for (final c in _removed) {
      c.dispose();
    }
    super.dispose();
  }

  Future<void> _loadPreset(String id) async {
    try {
      final item = await ref.read(itemRepositoryProvider).fetchItem(id);
      if (!mounted) return;
      setState(() => _addLine(item));
    } catch (_) {
      // If the item cannot load, the user can still add items manually.
    }
  }

  void _addLine(Item item) {
    _lines.putIfAbsent(item.id, () => _Line(item, _isManual ? '' : '1'));
  }

  void _removeLine(String id) {
    final line = _lines.remove(id);
    if (line != null) _removed.add(line.controller);
  }

  void _toggle(Item item) {
    setState(() {
      if (_lines.containsKey(item.id)) {
        _removeLine(item.id);
      } else {
        _addLine(item);
      }
    });
  }

  /// Stock change this line will cause.
  int _delta(_Line l) => switch (widget.type) {
    EntryType.supply => l.qty,
    EntryType.sale => -l.qty,
    EntryType.manual => l.qty,
  };

  /// Quick check on screen. The database does the final check.
  String? _lineError(_Line l) {
    final q = l.qty;
    switch (widget.type) {
      case EntryType.supply:
        if (q <= 0) return 'Enter a quantity above 0';
      case EntryType.sale:
        if (q <= 0) return 'Enter a quantity above 0';
        if (q > l.item.stock) return 'Only ${l.item.stock} in stock';
      case EntryType.manual:
        if (q == 0) return 'Enter an amount (+ or -)';
        if (l.item.stock + q < 0) {
          return 'Stock cannot go below 0 (now ${l.item.stock})';
        }
    }
    return null;
  }

  bool get _canSave => _lines.isNotEmpty && _lines.values.every((l) => _lineError(l) == null);

  void _step(_Line l, int by) {
    var next = l.qty + by;
    if (!_isManual && next < 1) next = 1;
    setState(() => l.controller.text = '$next');
  }

  Future<void> _pickDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: _date,
      firstDate: DateTime(2020),
      lastDate: _today(),
    );
    if (picked != null) {
      setState(() => _date = DateTime(picked.year, picked.month, picked.day));
    }
  }

  Future<void> _openPicker() {
    return showDialog<void>(
      context: context,
      builder: (_) => _ItemPickerDialog(
        type: widget.type,
        isAdded: (id) => _lines.containsKey(id),
        onToggle: _toggle,
      ),
    );
  }

  Future<void> _save() async {
    if (!(_formKey.currentState?.validate() ?? false)) return;
    final user = ref.read(authProvider);
    if (user == null || !_canSave) return;

    setState(() => _busy = true);
    try {
      await ref
          .read(inventoryRepositoryProvider)
          .createEntry(
            userId: user.id,
            type: widget.type,
            entryDate: _date,
            quantities: {for (final l in _lines.values) l.item.id: l.qty},
            referenceNo: _isManual ? null : _reference.text,
            note: _note.text,
          );
      refreshAll(ref);
      if (!mounted) return;
      showMessage(context, 'Saved');
      goBackOr(context, '/');
    } catch (e) {
      if (mounted) showMessage(context, errorMessage(e), isError: true);
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    final count = _lines.length;

    return Scaffold(
      appBar: AppBar(
        leading: IconButton(
          onPressed: () => goBackOr(context, '/'),
          icon: AppIcon(AppSvgs.chevronLeftIcon, color: AppColors.text),
        ),
        title: Text(_title),
      ),
      body: SingleChildScrollView(
        child: PageContainer(
          maxWidth: 760,
          child: Form(
            key: _formKey,
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                Text(_hint, style: const TextStyle(color: AppColors.textSoft)),
                const SizedBox(height: 16),
                if (!_isManual) ...[
                  TextFormField(
                    controller: _reference,
                    decoration: InputDecoration(
                      labelText: _isSale ? 'Voucher number' : 'Invoice / slip number (optional)',
                    ),
                    validator: _isSale ? (v) => (v == null || v.trim().isEmpty) ? 'Voucher number is required' : null : null,
                  ),
                  const SizedBox(height: 14),
                ],
                InkWell(
                  borderRadius: BorderRadius.circular(14),
                  onTap: _pickDate,
                  child: InputDecorator(
                    decoration: InputDecoration(
                      labelText: 'Date',
                      suffixIcon: Padding(
                        padding: const EdgeInsets.all(14),
                        child: AppIcon(
                          AppSvgs.calendarIcon,
                          size: 20,
                          color: AppColors.textSoft,
                        ),
                      ),
                    ),
                    child: Text(DateFormat('EEE, d MMM yyyy').format(_date)),
                  ),
                ),
                const SizedBox(height: 14),
                TextFormField(
                  controller: _note,
                  decoration: InputDecoration(
                    labelText: _isManual ? 'Reason (optional)' : 'Note (optional)',
                  ),
                ),
                const SizedBox(height: 24),
                Row(
                  children: [
                    const Expanded(
                      child: Text(
                        'Items',
                        style: TextStyle(fontSize: 18, fontWeight: FontWeight.w700),
                      ),
                    ),
                    OutlinedButton.icon(
                      onPressed: _openPicker,
                      icon: AppIcon(AppSvgs.addIcon, size: 18, color: AppColors.primary),
                      label: const Text('Add items'),
                    ),
                  ],
                ),
                const SizedBox(height: 12),
                if (_lines.isEmpty)
                  const AppCard(
                    child: Center(
                      child: Text(
                        'No items yet. Tap "Add items".',
                        style: TextStyle(color: AppColors.textSoft),
                      ),
                    ),
                  )
                else
                  for (final l in _lines.values)
                    Padding(
                      padding: const EdgeInsets.only(bottom: 10),
                      child: _lineCard(l),
                    ),
                const SizedBox(height: 16),
              ],
            ),
          ),
        ),
      ),
      bottomNavigationBar: SafeArea(
        child: Center(

          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 760),
            child: Padding(
              padding: const EdgeInsets.fromLTRB(16, 8, 16, 12),
              child: SizedBox(

                          child: CircularProgressIndicator(

  }

/// Tap items to add or remove them. The dialog stays open so you can add many.
class _ItemPickerDialog extends ConsumerStatefulWidget {
  const _ItemPickerDialog({
    required this.type,
    required this.isAdded,
    required this.onToggle,
  });

  final EntryType type;
  final bool Function(String id) isAdded;
  final void Function(Item item) onToggle;

  @override
  ConsumerState<_ItemPickerDialog> createState() => _ItemPickerDialogState();
}

class _ItemPickerDialogState extends ConsumerState<_ItemPickerDialog> {
  String _q = '';

  bool _matches(Item i) {
    if (_q.isEmpty) return true;
    final text = [
      i.name,
      i.itemSerial ?? '',
      i.brandName ?? '',
      i.categoryName ?? '',
      i.variation,
      i.size,
    ].join(' ').toLowerCase();
    return text.contains(_q);
  }

  @override
  Widget build(BuildContext context) {
    final async = ref.watch(itemsProvider);

    return Dialog(
      backgroundColor: Colors.white,
      child: ConstrainedBox(
        constraints: const BoxConstraints(maxWidth: 520, maxHeight: 640),
        child: Padding(
          padding: const EdgeInsets.all(16),
          child: Column(
            children: [
              Row(
                children: [
                  const Expanded(
                    child: Text(
                      'Select items',
                      style: TextStyle(fontSize: 18, fontWeight: FontWeight.w700),
                    ),
                  ),
                  TextButton(
                    onPressed: () => Navigator.pop(context),
                    child: const Text('Done'),
                  ),
                ],
              ),
              const SizedBox(height: 8),
              TextField(
                autofocus: true,
                onChanged: (v) => setState(() => _q = v.trim().toLowerCase()),
                decoration: InputDecoration(
                  hintText: 'Search items',
                  prefixIcon: Padding(
                    padding: const EdgeInsets.all(14),
                    child: AppIcon(
                      AppSvgs.searchIcon,
                      size: 20,
                      color: AppColors.textSoft,
                    ),
                  ),
                ),
              ),
              const SizedBox(height: 8),
              Expanded(
                child: async.when(
                  loading: () => const LoadingView(),
                  error: (e, _) => ErrorState(
                    error: e,
                    onRetry: () => ref.invalidate(itemsProvider),
                  ),
                  data: (items) {
                    final list = items.where(_matches).toList();
                    if (list.isEmpty) {
                      return const EmptyState(message: 'No items found.');
                    }
                    return ListView.builder(
                      itemCount: list.length,
                      itemBuilder: (context, i) {
                        final item = list[i];
                        final added = widget.isAdded(item.id);
                        final blocked = widget.type == EntryType.sale && item.stock == 0;
                        final sub = [item.subtitle, item.brandName ?? ''].where((s) => s.isNotEmpty).join(' · ');

                        return ListTile(
                          enabled: !blocked || added,
                          leading: ItemThumb(url: item.imageUrl, size: 40),
                          title: Text(item.name),
                          subtitle: sub.isEmpty ? null : Text(sub),
                          trailing: Text(
                            added ? 'Added' : '${item.stock}',
                            style: TextStyle(
                              fontWeight: FontWeight.w600,
                              color: added ? AppColors.success : (item.stock == 0 ? AppColors.danger : AppColors.text),
                            ),
                          ),
                          onTap: () {
                            widget.onToggle(item);
                            setState(() {});
                          },
                        );
                      },
                    );
                  },
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

const _stockLabels = {
  _StockFilter.all: 'All',
  _StockFilter.inStock: 'In stock',
  _StockFilter.outOfStock: 'Out of stock',
};

const _sortLabels = {
  _Sort.nameAsc: 'Name A-Z',
  _Sort.nameDesc: 'Name Z-A',
  _Sort.stockLow: 'Stock low-high',
  _Sort.stockHigh: 'Stock high-low',
  _Sort.newest: 'Newest',
};

class ItemsPage extends ConsumerStatefulWidget {
  const ItemsPage({super.key});

  @override
  ConsumerState<ItemsPage> createState() => _ItemsPageState();
}

class _ItemsPageState extends ConsumerState<ItemsPage> {
  String _query = '';
  String _brandId = ''; // '' means all
  String _categoryId = '';
  _StockFilter _stock = _StockFilter.all;
  _Sort _sort = _Sort.nameAsc;
  _ViewMode _view = _ViewMode.grid;

  /// Search, filter, and sort on the device. Fine for a bar-sized item list.
  List<Item> _apply(List<Item> items) {
    final q = _query.trim().toLowerCase();
    final list = items.where((i) {
      if (_brandId.isNotEmpty && i.brandId != _brandId) return false;
      if (_categoryId.isNotEmpty && i.categoryId != _categoryId) return false;
      if (_stock == _StockFilter.inStock && i.stock == 0) return false;
      if (_stock == _StockFilter.outOfStock && i.stock != 0) return false;
      if (q.isEmpty) return true;
      final text = [
        i.name,
        i.itemSerial ?? '',
        i.brandName ?? '',
        i.categoryName ?? '',
        i.variation,
        i.size,
        i.shortDesc,
      ].join(' ').toLowerCase();
      return text.contains(q);
    }).toList();

    int byName(Item a, Item b) => a.name.toLowerCase().compareTo(b.name.toLowerCase());

    switch (_sort) {
      case _Sort.nameAsc:
        list.sort(byName);
      case _Sort.nameDesc:
        list.sort((a, b) => byName(b, a));
      case _Sort.stockLow:
        list.sort((a, b) => a.stock.compareTo(b.stock));
      case _Sort.stockHigh:
        list.sort((a, b) => b.stock.compareTo(a.stock));
      case _Sort.newest:
        list.sort((a, b) => b.createdAt.compareTo(a.createdAt));
    }
    return list;
  }

  /// Brands or categories that exist in the item list. '' = All.
  Map<String, String> _lookupOptions(List<Item> items, {required bool brand}) {
    final found = <String, String>{};
    for (final i in items) {
      final id = brand ? i.brandId : i.categoryId;
      final name = brand ? i.brandName : i.categoryName;
      if (id != null) found[id] = name ?? '';
    }
    final sorted = found.entries.toList()..sort((a, b) => a.value.toLowerCase().compareTo(b.value.toLowerCase()));
    return {'': 'All', for (final e in sorted) e.key: e.value};
  }

  @override
  Widget build(BuildContext context) {
    final async = ref.watch(itemsProvider);

    return PageContainer(
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          PageHeader(
            title: 'Items',
            subtitle: async.hasValue ? '${async.requireValue.length} items' : null,
            trailing: FilledButton.icon(
              onPressed: () => context.push('/items/new'),
              icon: AppIcon(AppSvgs.addIcon, size: 18, color: Colors.white),
              label: const Text('New item'),
            ),
          ),
          const SizedBox(height: 16),
          TextField(
            onChanged: (v) => setState(() => _query = v),
            decoration: InputDecoration(
              hintText: 'Search name, serial, brand...',
              prefixIcon: Padding(
                padding: const EdgeInsets.all(14),
                child: AppIcon(
                  AppSvgs.searchIcon,
                  size: 20,
                  color: AppColors.textSoft,
                ),
              ),
            ),
          ),
          const SizedBox(height: 12),
          async.when(
            loading: () => const Expanded(child: LoadingView()),
            error: (e, _) => Expanded(
              child: ErrorState(
                error: e,
                onRetry: () => ref.invalidate(itemsProvider),
              ),
            ),
            data: (items) {
              final shown = _apply(items);
              return Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.stretch,
                  children: [
                    _controls(items),
                    const SizedBox(height: 14),
                    Expanded(
                      child: items.isEmpty
                          ? EmptyState(
                              message: 'No items yet. Create your first item.',
                              action: FilledButton(
                                onPressed: () => context.push('/items/new'),
                                child: const Text('New item'),
                              ),
                            )
                          : shown.isEmpty
                          ? const EmptyState(
                              message: 'No items match your search.',
                            )
                          : switch (_view) {
                              _ViewMode.grid => _ItemGrid(items: shown),
                              _ViewMode.list => _ItemList(items: shown),
                              _ViewMode.table => _ItemTable(items: shown),
                            },
                    ),
                  ],
                ),
              );
            },
          ),
        ],
      ),
    );
  }

  Widget _controls(List<Item> items) {
    return SingleChildScrollView(
      scrollDirection: Axis.horizontal,
      child: Row(
        children: [
          _MenuChip<String>(
            label: 'Brand',
            icon: AppSvgs.filterIcon,
            value: _brandId,
            options: _lookupOptions(items, brand: true),
            onSelected: (v) => setState(() => _brandId = v),
          ),
          const SizedBox(width: 8),
          _MenuChip<String>(
            label: 'Category',
            icon: AppSvgs.filterIcon,
            value: _categoryId,
            options: _lookupOptions(items, brand: false),
            onSelected: (v) => setState(() => _categoryId = v),
          ),
          const SizedBox(width: 8),
          _MenuChip<_StockFilter>(
            label: 'Stock',
            value: _stock,
            options: _stockLabels,
            onSelected: (v) => setState(() => _stock = v),
          ),
          const SizedBox(width: 8),
          _MenuChip<_Sort>(
            label: 'Sort',
            icon: AppSvgs.sortIcon,
            value: _sort,
            options: _sortLabels,
            onSelected: (v) => setState(() => _sort = v),
          ),
          const SizedBox(width: 12),
          _ViewToggle(
            value: _view,
            onChanged: (v) => setState(() => _view = v),
          ),
        ],
      ),
    );
  }
}

/// A pill button that opens a small menu.
class _MenuChip<T extends Object> extends StatelessWidget {
  const _MenuChip({
    required this.label,
    required this.value,
    required this.options,
    required this.onSelected,
    this.icon,
  });

  final String label;
  final String? icon;
  final T value;
  final Map<T, String> options;
  final ValueChanged<T> onSelected;

  @override
  Widget build(BuildContext context) {
    return PopupMenuButton<T>(
      tooltip: label,
      onSelected: onSelected,
      itemBuilder: (_) => [
        for (final e in options.entries) PopupMenuItem<T>(value: e.key, child: Text(e.value)),
      ],
      child: Container(
        padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 10),
        decoration: BoxDecoration(
          color: Colors.white,
          borderRadius: BorderRadius.circular(20),
          border: Border.all(color: AppColors.border),
        ),
        child: Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            if (icon != null) ...[
              AppIcon(icon!, size: 16, color: AppColors.textSoft),
              const SizedBox(width: 8),
            ],
            Text(
              '$label: ${options[value] ?? 'All'}',
              style: const TextStyle(fontSize: 13, fontWeight: FontWeight.w500),
            ),
            const SizedBox(width: 4),
            RotatedBox(
              quarterTurns: 1,
              child: AppIcon(
                AppSvgs.chevronRightIcon,
                size: 14,
                color: AppColors.textSoft,
              ),
            ),
          ],
        ),
      ),
    );
  }
}

class ItemFormPage extends ConsumerStatefulWidget {
  const ItemFormPage({super.key, this.itemId});

  /// null = create, otherwise edit this item.
  final String? itemId;

  @override
  ConsumerState<ItemFormPage> createState() => _ItemFormPageState();
}

class _ItemFormPageState extends ConsumerState<ItemFormPage> {
  final _formKey = GlobalKey<FormState>();
  final _serial = TextEditingController();
  final _name = TextEditingController();
  final _image = TextEditingController();
  final _desc = TextEditingController();
  final _size = TextEditingController();
  final _variation = TextEditingController();

  Lookup? _brand;
  Lookup? _category;
  bool _loading = false;
  bool _busy = false;
  Object? _loadError;

  bool get _isEdit => widget.itemId != null;

  @override
  void initState() {
    super.initState();
    _loading = _isEdit;
    if (_isEdit) _load();
  }

  @override
  void dispose() {
    for (final c in [_serial, _name, _image, _desc, _size, _variation]) {
      c.dispose();
    }
    super.dispose();
  }

  Future<void> _load() async {
    try {
      final item = await ref.read(itemRepositoryProvider).fetchItem(widget.itemId!);
      if (!mounted) return;
      setState(() {
        _serial.text = item.itemSerial ?? '';
        _name.text = item.name;
        _image.text = item.imageUrl ?? '';
        _desc.text = item.shortDesc;
        _size.text = item.size;
        _variation.text = item.variation;
        _brand = item.brandId == null ? null : Lookup(id: item.brandId!, name: item.brandName ?? '');
        _category = item.categoryId == null ? null : Lookup(id: item.categoryId!, name: item.categoryName ?? '');
        _loading = false;
      });
    } catch (e) {
      if (mounted) {
        setState(() {
          _loadError = e;
          _loading = false;
        });
      }
    }
  }

  Future<_PickResult?> _openLookup({
    required String noun,
    required Future<List<Lookup>> Function() load,
    required Future<Lookup> Function(String) create,
    required String? currentId,
    required bool startInCreateMode,
  }) async {
    try {
      final options = await load();
      if (!mounted) return null;
      return showDialog<_PickResult>(
        context: context,
        builder: (_) => _LookupDialog(
          noun: noun,
          options: options,
          create: create,
          currentId: currentId,
          startInCreateMode: startInCreateMode,
        ),
      );
    } catch (e) {
      if (mounted) showMessage(context, errorMessage(e), isError: true);
      return null;
    }
  }

  Future<void> _pickBrand({bool create = false}) async {
    final repo = ref.read(itemRepositoryProvider);
    final r = await _openLookup(
      noun: 'brand',
      load: repo.fetchBrands,
      create: repo.createBrand,
      currentId: _brand?.id,
      startInCreateMode: create,
    );
    if (r != null) {
      setState(() => _brand = r.lookup);
      ref.invalidate(brandsProvider);
    }
  }

  Future<void> _pickCategory({bool create = false}) async {
    final repo = ref.read(itemRepositoryProvider);
    final r = await _openLookup(
      noun: 'category',
      load: repo.fetchCategories,
      create: repo.createCategory,
      currentId: _category?.id,
      startInCreateMode: create,
    );
    if (r != null) {
      setState(() => _category = r.lookup);
      ref.invalidate(categoriesProvider);
    }
  }

  Future<void> _save() async {
    if (!(_formKey.currentState?.validate() ?? false)) return;
    final user = ref.read(authProvider);
    if (user == null) return;

    setState(() => _busy = true);
    try {
      final input = ItemInput(
        itemSerial: _serial.text,
        name: _name.text,
        brandId: _brand?.id,
        categoryId: _category?.id,
        imageUrl: _image.text,
        shortDesc: _desc.text,
        size: _size.text,
        variation: _variation.text,
      );
      final repo = ref.read(itemRepositoryProvider);
      if (_isEdit) {
        await repo.updateItem(widget.itemId!, user.id, input);
      } else {
        await repo.createItem(user.id, input);
      }
      refreshAll(ref);
      if (!mounted) return;
      showMessage(context, _isEdit ? 'Item updated' : 'Item created');
      goBackOr(context, '/items');
    } catch (e) {
      if (!mounted) return;
      final msg = (e is PostgrestException && e.code == '23505') ? 'This item serial is already used.' : errorMessage(e);
      showMessage(context, msg, isError: true);
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  String? _required(String? v, String what) => (v == null || v.trim().isEmpty) ? '$what is required' : null;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        leading: IconButton(
          onPressed: () => goBackOr(context, '/items'),
          icon: AppIcon(AppSvgs.chevronLeftIcon, color: AppColors.text),
        ),
        title: Text(_isEdit ? 'Edit item' : 'New item'),
      ),
      body: _loadError != null
          ? ErrorState(error: _loadError!)
          : _loading
          ? const LoadingView()
          : SingleChildScrollView(
              child: PageContainer(
                maxWidth: 640,
                child: Form(
                  key: _formKey,
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.stretch,
                    children: [
                      Row(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          ItemThumb(url: _image.text, size: 72),
                          const SizedBox(width: 14),
                          Expanded(
                            child: TextFormField(
                              controller: _image,
                              keyboardType: TextInputType.url,
                              decoration: const InputDecoration(
                                labelText: 'Image URL (optional)',
                              ),
                              onChanged: (_) => setState(() {}),
                              validator: (v) {
                                final t = (v ?? '').trim().toLowerCase();
                                if (t.isEmpty) return null;
                                return t.startsWith('http') ? null : 'Must start with http:// or https://';
                              },
                            ),
                          ),
                        ],
                      ),
                      const SizedBox(height: 14),
                      TextFormField(
                        controller: _serial,
                        decoration: const InputDecoration(labelText: 'Item serial (optional)'),
                      ),
                      const SizedBox(height: 14),
                      TextFormField(
                        controller: _name,
                        textCapitalization: TextCapitalization.words,
                        decoration: const InputDecoration(labelText: 'Name'),
                        validator: (v) => _required(v, 'Name'),
                      ),
                      const SizedBox(height: 14),
                      _LookupRow(
                        label: 'Brand',
                        value: _brand?.name,
                        onPick: _pickBrand,
                        onAdd: () => _pickBrand(create: true),
                      ),
                      const SizedBox(height: 14),
                      _LookupRow(
                        label: 'Category',
                        value: _category?.name,
                        onPick: _pickCategory,
                        onAdd: () => _pickCategory(create: true),
                      ),
                      const SizedBox(height: 14),
                      Row(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          Expanded(
                            child: TextFormField(
                              controller: _size,
                              decoration: const InputDecoration(
                                labelText: 'Size',
                                hintText: '1L, 350g, 500ml, 1pcs',
                              ),
                            ),
                          ),
                          const SizedBox(width: 12),
                          Expanded(
                            child: TextFormField(
                              controller: _variation,
                              decoration: const InputDecoration(
                                labelText: 'Variation',
                                hintText: 'strawberry, menthol',
                              ),
                            ),
                          ),
                        ],
                      ),
                      const SizedBox(height: 14),
                      TextFormField(
                        controller: _desc,
                        maxLines: 2,
                        decoration: const InputDecoration(
                          labelText: 'Short description (optional)',
                        ),
                      ),
                      const SizedBox(height: 24),
                      FilledButton(
                        onPressed: _busy ? null : _save,
                        child: _busy
                            ? const SizedBox(
                                width: 20,
                                height: 20,
                                child: CircularProgressIndicator(
                                  strokeWidth: 2,
                                  color: Colors.white,
                                ),
                              )
                            : Text(_isEdit ? 'Save changes' : 'Create item'),
                      ),
                      const SizedBox(height: 24),
                    ],
                  ),
                ),
              ),
            ),
    );
  }
}

/// Pick an existing brand/category, or create a new one.
/// Creating a name that already exists just selects the existing one.
class _LookupDialog extends StatefulWidget {
  const _LookupDialog({
    required this.noun,
    required this.options,
    required this.create,
    required this.currentId,
    this.startInCreateMode = false,
  });

  /// "brand" or "category"
  final String noun;
  final List<Lookup> options;
  final Future<Lookup> Function(String name) create;
  final String? currentId;
  final bool startInCreateMode;

  @override
  State<_LookupDialog> createState() => _LookupDialogState();
}

class _LookupDialogState extends State<_LookupDialog> {
  final _search = TextEditingController();
  final _newName = TextEditingController();
  late bool _creating = widget.startInCreateMode;
  String? _error;
  bool _busy = false;

  @override
  void dispose() {
    _search.dispose();
    _newName.dispose();
    super.dispose();
  }

  String get _noun => widget.noun;
  String get _nounTitle => _noun[0].toUpperCase() + _noun.substring(1);

  List<Lookup> get _filtered {
    final q = _search.text.trim().toLowerCase();
    return widget.options.where((o) => q.isEmpty || o.name.toLowerCase().contains(q)).toList();
  }

  void _startCreate() {
    setState(() {
      _newName.text = _search.text.trim();
      _error = null;
      _creating = true;
    });
  }

  Future<void> _create() async {
    final name = _newName.text.trim();
    if (name.isEmpty) {
      setState(() => _error = 'Enter a name');
      return;
    }

    // Already exists? Just select it.
    for (final o in widget.options) {
      if (o.name.toLowerCase() == name.toLowerCase()) {
        Navigator.pop(context, _PickResult(o));
        return;
      }
    }

    setState(() {
      _busy = true;
      _error = null;
    });
    try {
      final created = await widget.create(name);
      if (mounted) Navigator.pop(context, _PickResult(created));
    } catch (e) {
      if (mounted) {
        setState(() {
          _error = errorMessage(e);
          _busy = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return _creating ? _createView() : _listView();
  }

  Widget _createView() {
    return AlertDialog(
      backgroundColor: Colors.white,
      title: Text('New $_noun'),
      content: SizedBox(
        width: 360,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            TextField(
              controller: _newName,
              autofocus: true,
              enabled: !_busy,
              textCapitalization: TextCapitalization.words,
              onSubmitted: (_) => _create(),
              decoration: InputDecoration(labelText: '$_nounTitle name'),
            ),
            if (_error != null)
              Padding(
                padding: const EdgeInsets.only(top: 8),
                child: Text(
                  _error!,
                  style: const TextStyle(color: AppColors.danger),
                ),
              ),
          ],
        ),
      ),
      actions: [
        TextButton(
          onPressed: _busy
              ? null
              : () {
                  if (widget.startInCreateMode) {
                    Navigator.pop(context);
                  } else {
                    setState(() {
                      _creating = false;
                      _error = null;
                    });
                  }
                },
          child: Text(widget.startInCreateMode ? 'Cancel' : 'Back'),
        ),
        FilledButton(
          onPressed: _busy ? null : _create,
          child: _busy
              ? const SizedBox(
                  width: 18,
                  height: 18,
                  child: CircularProgressIndicator(
                    strokeWidth: 2,
                    color: Colors.white,
                  ),
                )
              : Text('Add $_noun'),
        ),
      ],
    );
  }

  Widget _listView() {
    final list = _filtered;
    return AlertDialog(
      backgroundColor: Colors.white,
      title: Text('Select $_noun'),
      content: SizedBox(
        width: 360,
        height: 400,
        child: Column(
          children: [
            TextField(
              controller: _search,
              autofocus: true,
              onChanged: (_) => setState(() {}),
              decoration: InputDecoration(hintText: 'Search $_noun'),
            ),
            const SizedBox(height: 8),
            Expanded(
              child: ListView(
                children: [
                  ListTile(
                    leading: AppIcon(AppSvgs.addIcon, color: AppColors.primary),
                    title: Text(
                      'New $_noun',
                      style: const TextStyle(
                        color: AppColors.primary,
                        fontWeight: FontWeight.w600,
                      ),
                    ),
                    onTap: _startCreate,
                  ),
                  ListTile(
                    title: const Text('None'),
                    onTap: () => Navigator.pop(context, const _PickResult(null)),
                  ),
                  for (final o in list)
                    ListTile(
                      title: Text(o.name),
                      selected: o.id == widget.currentId,
                      selectedColor: AppColors.primary,
                      onTap: () => Navigator.pop(context, _PickResult(o)),
                    ),
                  if (list.isEmpty)
                    Padding(
                      padding: const EdgeInsets.all(16),
                      child: Text(
                        'No $_noun found. Tap "New $_noun" to add one.',
                        textAlign: TextAlign.center,
                        style: const TextStyle(color: AppColors.textSoft),
                      ),
                    ),
                ],
              ),
            ),
          ],
        ),
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.pop(context),
          child: const Text('Cancel'),
        ),
      ],
    );
  }
}

class ItemDetailPage extends ConsumerWidget {
  const ItemDetailPage({super.key, required this.itemId});

  final String itemId;

  Future<void> _archive(BuildContext context, WidgetRef ref, Item item) async {
    final ok = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        backgroundColor: Colors.white,
        title: const Text('Delete item?'),
        content: Text(
          '"${item.name}" will be removed from the item list.'
          '${item.stock > 0 ? ' It still has ${item.stock} in stock.' : ''}'

            style: TextButton.styleFrom(foregroundColor: AppColors.danger),
            child: const Text('Delete'),
          ),
        ],
      ),
    );

    if (user == null) return;

import 'package:flutter/material.dart';

DateTime _dateOnly(DateTime d) => DateTime(d.year, d.month, d.day);
DateTime _addDays(DateTime d, int n) => DateTime(d.year, d.month, d.day + n);

int _daysBetween(DateTime a, DateTime b) => DateTime.utc(b.year, b.month, b.day).difference(DateTime.utc(a.year, a.month, a.day)).inDays;

enum _Preset { today, yesterday, last7, thisMonth, custom }

class ReportsPage extends ConsumerStatefulWidget {
  const ReportsPage({super.key});

  @override
  ConsumerState<ReportsPage> createState() => _ReportsPageState();
}

class _ReportsPageState extends ConsumerState<ReportsPage> {
  static const _maxDays = 93;
  static final _dayFormat = DateFormat('d MMM yyyy');

  _Preset _preset = _Preset.today;
  late DateTime _from = _dateOnly(DateTime.now());
  late DateTime _to = _from;
  int _sheetIndex = 0;
  bool _onlyMovement = false;

  String get _rangeLabel => _from == _to ? DateFormat('EEE, d MMM yyyy').format(_from) : '${_dayFormat.format(_from)} - ${_dayFormat.format(_to)}';

  void _applyPreset(_Preset p) {
    final today = _dateOnly(DateTime.now());
    final (from, to) = switch (p) {
      _Preset.today => (today, today),
      _Preset.yesterday => (_addDays(today, -1), _addDays(today, -1)),
      _Preset.last7 => (_addDays(today, -6), today),
      _Preset.thisMonth => (DateTime(today.year, today.month, 1), today),
      _Preset.custom => (_from, _to),
    };
    setState(() {
      _preset = p;
      _from = from;
      _to = to;
      _sheetIndex = 0;
    });
  }

  Future<void> _pickRange() async {
    final today = _dateOnly(DateTime.now());
    final picked = await showDateRangePicker(
      context: context,
      firstDate: DateTime(2020),
      lastDate: today,
      initialDateRange: DateTimeRange(start: _from, end: _to),
      helpText: 'Select report dates',
    );
    if (picked == null) return;

    final from = _dateOnly(picked.start);
    final to = _dateOnly(picked.end);

    if (_daysBetween(from, to) + 1 > _maxDays) {
      if (mounted) {
        showMessage(context, 'Please choose $_maxDays days or fewer.', isError: true);
      }
      return;
    }

    setState(() {
      _preset = _Preset.custom;
      _from = from;
      _to = to;
      _sheetIndex = 0;
    });
  }

  void _export(List<ReportSheet> sheets) {
    try {
      ReportExporter.exportAndDownload(sheets);
      showMessage(
        context,
        sheets.length == 1 ? 'Excel file downloaded' : 'Excel file with ${sheets.length} sheets downloaded',
      );
    } catch (e) {
      showMessage(context, errorMessage(e), isError: true);
    }
  }

  @override
  Widget build(BuildContext context) {
    final query = (from: _from, to: _to);
    final async = ref.watch(stockReportProvider(query));
    final mobile = MediaQuery.sizeOf(context).width < 700;

    return PageContainer(
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          PageHeader(
            title: 'Reports',
            subtitle: _rangeLabel,
            trailing: FilledButton.icon(
              onPressed: async.hasValue ? () => _export(async.requireValue) : null,
              icon: AppIcon(AppSvgs.downloadIcon, size: 18, color: Colors.white),
              label: Text(mobile ? 'Export' : 'Export Excel'),
            ),
          ),
          const SizedBox(height: 14),
          _presetChips(),
          const SizedBox(height: 14),
          Expanded(
            child: async.when(
              loading: () => const LoadingView(),
              error: (e, _) => ErrorState(
                error: e,
                onRetry: () => ref.invalidate(stockReportProvider(query)),
              ),
              data: _content,
            ),
          ),
        ],
      ),
    );
  }

  Widget _chip({
    required String label,
    required bool selected,
    required VoidCallback onTap,
    Widget? avatar,
  }) {
    return Padding(
      padding: const EdgeInsets.only(right: 8),
      child: ChoiceChip(
        label: Text(label),
        avatar: avatar,
        selected: selected,
        showCheckmark: false,
        selectedColor: AppColors.primarySoft,
        backgroundColor: Colors.white,
        side: BorderSide(color: selected ? AppColors.primary : AppColors.border),
        onSelected: (_) => onTap(),
      ),
    );
  }

  Widget _presetChips() {
    const presets = <(_Preset, String)>[
      (_Preset.today, 'Today'),
      (_Preset.yesterday, 'Yesterday'),
      (_Preset.last7, 'Last 7 days'),
      (_Preset.thisMonth, 'This month'),
    ];

    return SingleChildScrollView(
      scrollDirection: Axis.horizontal,
      child: Row(
        children: [
          for (final (preset, label) in presets)
            _chip(
              label: label,
              selected: _preset == preset,
              onTap: () => _applyPreset(preset),
            ),
          _chip(
            label: _preset == _Preset.custom ? _rangeLabel : 'Pick dates',
            selected: _preset == _Preset.custom,
            onTap: _pickRange,
            avatar: AppIcon(
              AppSvgs.calendarIcon,
              size: 16,
              color: _preset == _Preset.custom ? AppColors.primary : AppColors.textSoft,
            ),
          ),
        ],
      ),
    );
  }

  Widget _content(List<ReportSheet> sheets) {
    final index = math.min(_sheetIndex, sheets.length - 1);
    final sheet = sheets[index];
    final rows = _onlyMovement ? sheet.rows.where((r) => r.hasMovement).toList() : sheet.rows;

    return Column(
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        // Totals of the selected date
        Row(
          children: [
            Expanded(child: _MiniStat('Opening', sheet.totalOpening, AppColors.text)),
            const SizedBox(width: 8),
            Expanded(child: _MiniStat('Supplied', sheet.totalSupply, AppColors.success)),
            const SizedBox(width: 8),
            Expanded(child: _MiniStat('Sold', sheet.totalSale, AppColors.primary)),
            const SizedBox(width: 8),
            Expanded(child: _MiniStat('Closing', sheet.totalClosing, AppColors.text)),
          ],
        ),
        const SizedBox(height: 10),
        Row(
          children: [
            Expanded(
              child: Text(
                DateFormat('EEEE, d MMM yyyy').format(sheet.date),
                style: const TextStyle(fontSize: 16, fontWeight: FontWeight.w700),
              ),
            ),
            FilterChip(
              label: const Text('With movement'),
              selected: _onlyMovement,
              showCheckmark: false,
              selectedColor: AppColors.primarySoft,
              backgroundColor: Colors.white,
              side: BorderSide(color: _onlyMovement ? AppColors.primary : AppColors.border),
              onSelected: (v) => setState(() => _onlyMovement = v),
            ),
          ],
        ),
        const SizedBox(height: 10),
        Expanded(
          child: rows.isEmpty
              ? EmptyState(
                  message: _onlyMovement ? 'No stock movement on this day.' : 'No items yet.',
                )
              : _SheetTable(
                  key: ValueKey(sheet.date),
                  sheet: sheet,
                  rows: rows,
                  filtered: _onlyMovement,
                ),
        ),
        if (sheets.length > 1) ...[
          const SizedBox(height: 10),
          _SheetTabs(
            sheets: sheets,
            selected: index,
            onSelect: (i) => setState(() => _sheetIndex = i),
          ),
        ],
      ],
    );
  }
}

class _SheetTable extends StatefulWidget {
  const _SheetTable({
    super.key,
    required this.sheet,
    required this.rows,
    required this.filtered,
  });

  final ReportSheet sheet;
  final List<ReportRow> rows;

  /// True when some items are hidden. The total row still counts all items.
  final bool filtered;

  @override
  State<_SheetTable> createState() => _SheetTableState();
}

class _SheetTableState extends State<_SheetTable> {
  static const double _rowH = 52;
  static const double _headH = 46;

  final _head = ScrollController();
  final _body = ScrollController();

  @override
  void initState() {
    super.initState();
    _body.addListener(_syncHeader);
  }

  /// Keeps the header columns in line with the body when scrolling sideways.
  void _syncHeader() {
    if (!_head.hasClients) return;
    final target = _body.offset.clamp(0.0, _head.position.maxScrollExtent).toDouble();
    if ((_head.offset - target).abs() > 0.5) _head.jumpTo(target);
  }

  @override
  void dispose() {
    _body.removeListener(_syncHeader);
    _head.dispose();
    _body.dispose();
    super.dispose();
  }

  Widget _cell(
    String text,
    double width, {
    double height = _rowH,
    bool right = false,
    bool bold = false,
    Color? color,
    Color? bg,
    int maxLines = 1,
    double fontSize = 13,
  }) {
    return Container(
      width: width,
      height: height,
      alignment: right ? Alignment.centerRight : Alignment.centerLeft,
      padding: const EdgeInsets.symmetric(horizontal: 10),
      decoration: BoxDecoration(
        color: bg,
        border: const Border(
          right: BorderSide(color: AppColors.border),
          bottom: BorderSide(color: AppColors.border),
        ),
      ),
      child: Text(
        text,
        maxLines: maxLines,
        overflow: TextOverflow.ellipsis,
        textAlign: right ? TextAlign.right : TextAlign.left,
        style: TextStyle(
          fontSize: fontSize,
          height: 1.2,
          fontWeight: bold ? FontWeight.w700 : FontWeight.w500,
          color: color ?? AppColors.text,
        ),
      ),
    );
  }

  Widget _num(int v, double width, {Color? hot, bool bold = false, Color? bg}) {
    return _cell(
      '$v',
      width,
      right: true,
      bold: bold,
      bg: bg,
      color: v == 0 ? AppColors.textSoft : (hot ?? AppColors.text),
    );
  }

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, c) {
        final mobile = c.maxWidth < 700;
        final nameW = mobile ? 150.0 : 300.0;
        final numW = mobile ? 78.0 : 112.0;
        final remarkMin = mobile ? 190.0 : 280.0;
        final rightW = math.max(c.maxWidth - nameW - 2, numW * 4 + remarkMin);
        final remarkW = rightW - numW * 4;

        const head = AppColors.muted;
        final s = widget.sheet;

        Widget headerRow() => Row(
          children: [
            _cell('Opening Stock', numW, height: _headH, right: true, bold: true, bg: head, fontSize: 12),
            _cell('Supply Qty', numW, height: _headH, right: true, bold: true, bg: head, fontSize: 12),
            _cell('Sale Qty', numW, height: _headH, right: true, bold: true, bg: head, fontSize: 12),
            _cell('Closing Stock', numW, height: _headH, right: true, bold: true, bg: head, fontSize: 12),
            _cell('Remark', remarkW, height: _headH, bold: true, bg: head, fontSize: 12),
          ],
        );

        Widget dataRow(ReportRow r) => Row(
          children: [
            _num(r.opening, numW),
            _num(r.supply, numW, hot: AppColors.success),
            _num(r.sale, numW, hot: AppColors.primary),
            _num(r.closing, numW, bold: true),
            _cell(r.remark, remarkW, maxLines: 2, color: AppColors.textSoft, fontSize: 12),
          ],
        );

        Widget totalRow() => Row(
          children: [
            _num(s.totalOpening, numW, bold: true, bg: head),
            _num(s.totalSupply, numW, hot: AppColors.success, bold: true, bg: head),
            _num(s.totalSale, numW, hot: AppColors.primary, bold: true, bg: head),
            _num(s.totalClosing, numW, bold: true, bg: head),
            _cell('', remarkW, bg: head),
          ],
        );

        return Container(
          clipBehavior: Clip.antiAlias,
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(16),
            border: Border.all(color: AppColors.border),
          ),
          child: Column(
            children: [
              // Header (stays on top)
              Row(
                children: [
                  _cell('Item Name', nameW, height: _headH, bold: true, bg: head, fontSize: 12),
                  Expanded(
                    child: SingleChildScrollView(
                      controller: _head,
                      scrollDirection: Axis.horizontal,
                      physics: const NeverScrollableScrollPhysics(),
                      child: headerRow(),
                    ),
                  ),
                ],
              ),
              // Body
              Expanded(
                child: SingleChildScrollView(
                  child: Row(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      // Item names stay in place when scrolling sideways
                      Column(
                        children: [
                          for (final r in widget.rows) _cell(r.displayName, nameW, maxLines: 2),
                          _cell(
                            widget.filtered ? 'Total (all items)' : 'Total',
                            nameW,
                            bold: true,
                            bg: head,
                          ),
                        ],
                      ),
                      Expanded(
                        child: SingleChildScrollView(
                          controller: _body,
                          scrollDirection: Axis.horizontal,
                          child: Column(
                            children: [
                              for (final r in widget.rows) dataRow(r),
                              totalRow(),
                            ],
                          ),
                        ),
                      ),
                    ],
                  ),
                ),
              ),
            ],
          ),
        );
      },
    );
  }
}

/// Sheet tabs at the bottom, like Excel. One tab per date.
class _SheetTabs extends StatefulWidget {
  const _SheetTabs({
    required this.sheets,
    required this.selected,
    required this.onSelect,
  });

  final List<ReportSheet> sheets;
  final int selected;
  final ValueChanged<int> onSelect;

  @override
  State<_SheetTabs> createState() => _SheetTabsState();
}

class _SheetTabsState extends State<_SheetTabs> {
  static const double _tabW = 84;
  final _ctrl = ScrollController();

  @override
  void didUpdateWidget(covariant _SheetTabs oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.selected != widget.selected) _reveal();
  }

  /// Scrolls so the selected tab is in the middle.
  void _reveal() {
    if (!_ctrl.hasClients) return;
    final pos = _ctrl.position;
    final target = widget.selected * _tabW - (pos.viewportDimension - _tabW) / 2;
    _ctrl.animateTo(
      target.clamp(0.0, pos.maxScrollExtent).toDouble(),
      duration: const Duration(milliseconds: 200),
      curve: Curves.easeOut,
    );
  }

  @override
  void dispose() {
    _ctrl.dispose();
    super.dispose();
  }

  Widget _arrow(String icon, bool enabled, VoidCallback onTap) {
    return IconButton(
      visualDensity: VisualDensity.compact,
      onPressed: enabled ? onTap : null,
      icon: AppIcon(icon, size: 18, color: enabled ? AppColors.text : AppColors.border),
    );
  }

  @override
  Widget build(BuildContext context) {
    final last = widget.sheets.length - 1;

    return Container(
      height: 52,
      decoration: BoxDecoration(
        color: AppColors.muted,
        borderRadius: BorderRadius.circular(14),
        border: Border.all(color: AppColors.border),
      ),
      child: Row(
        children: [
          _arrow(
            AppSvgs.chevronLeftIcon,
            widget.selected > 0,
            () => widget.onSelect(widget.selected - 1),
          ),
          Expanded(
            child: ListView.builder(
              controller: _ctrl,
              scrollDirection: Axis.horizontal,
              itemCount: widget.sheets.length,
              itemBuilder: (context, i) {
                final isSelected = i == widget.selected;
                final date = widget.sheets[i].date;
                return SizedBox(
                  width: _tabW,
                  child: Padding(
                    padding: const EdgeInsets.symmetric(horizontal: 3, vertical: 6),
                    child: Material(
                      color: isSelected ? Colors.white : Colors.transparent,
                      shape: RoundedRectangleBorder(
                        borderRadius: BorderRadius.circular(10),
                        side: BorderSide(
                          color: isSelected ? AppColors.primary : Colors.transparent,
                        ),
                      ),
                      child: InkWell(
                        borderRadius: BorderRadius.circular(10),
                        onTap: () => widget.onSelect(i),
                        child: Column(
                          mainAxisAlignment: MainAxisAlignment.center,
                          children: [
                            Text(
                              DateFormat('d MMM').format(date),
                              style: TextStyle(
                                fontSize: 13,
                                height: 1.2,
                                fontWeight: isSelected ? FontWeight.w700 : FontWeight.w500,
                                color: isSelected ? AppColors.primary : AppColors.text,
                              ),
                            ),
                            Text(
                              DateFormat('EEE').format(date),
                              style: const TextStyle(
                                fontSize: 10,
                                height: 1.2,
                                color: AppColors.textSoft,
                              ),
                            ),
                          ],
                        ),
                      ),
                    ),
                  ),
                );
              },
            ),
          ),
          _arrow(
            AppSvgs.chevronRightIcon,
            widget.selected < last,
            () => widget.onSelect(widget.selected + 1),
          ),
        ],
      ),
    );
  }
}

/// Builds an .xlsx file with one sheet per date and downloads it.
/// Google Sheets can open .xlsx files (Drive > Upload > Open with Google Sheets).
class ReportExporter {
  static const _headers = [
    'Item Name',
    'Opening Stock',
    'Supply Qty',
    'Sale Qty',
    'Closing Stock',
    'Remark',
  ];

  static const _mime = 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet';

  static void exportAndDownload(List<ReportSheet> sheets) {
    if (sheets.isEmpty) throw const AppException('Nothing to export.');

    final bytes = build(sheets);
    final f = DateFormat('yyyy-MM-dd');
    final first = f.format(sheets.first.date);
    final last = f.format(sheets.last.date);
    final name = sheets.length == 1 ? 'stock_report_$first.xlsx' : 'stock_report_${first}_to_$last.xlsx';

    downloadBytes(bytes, name, mimeType: _mime);
  }

  static Uint8List build(List<ReportSheet> sheets) {
    final excel = xl.Excel.createExcel();
    final defaultName = excel.tables.keys.first; // empty "Sheet1" made by the library
    final nameFormat = DateFormat('d MMM yyyy'); // e.g. "12 Oct 2026"

    for (final sheet in sheets) {
      _writeSheet(excel[nameFormat.format(sheet.date)], sheet);
    }
    excel.delete(defaultName);

    final bytes = excel.encode();
    if (bytes == null) throw const AppException('Could not create the Excel file.');
    return Uint8List.fromList(bytes);
  }

  static void _writeSheet(xl.Sheet s, ReportSheet sheet) {
    xl.CellIndex at(int col, int row) => xl.CellIndex.indexByColumnRow(columnIndex: col, rowIndex: row);

    final grey = xl.ExcelColor.fromHexString('#F3F3F5');
    final titleStyle = xl.CellStyle(bold: true, fontSize: 14);
    final headLeft = xl.CellStyle(bold: true, backgroundColorHex: grey);
    final headRight = xl.CellStyle(
      bold: true,
      backgroundColorHex: grey,
      horizontalAlign: xl.HorizontalAlign.Right,
    );
    final number = xl.CellStyle(horizontalAlign: xl.HorizontalAlign.Right);
    final totalLeft = xl.CellStyle(bold: true, backgroundColorHex: grey);
    final totalRight = xl.CellStyle(
      bold: true,
      backgroundColorHex: grey,
      horizontalAlign: xl.HorizontalAlign.Right,
    );

    // Row 1: title
    s.updateCell(
      at(0, 0),
      xl.TextCellValue('Stock report - ${DateFormat('EEEE, d MMM yyyy').format(sheet.date)}'),
      cellStyle: titleStyle,
    );

    // Row 3: column headers
    const headerRow = 2;
    for (var c = 0; c < _headers.length; c++) {
      s.updateCell(
        at(c, headerRow),
        xl.TextCellValue(_headers[c]),
        cellStyle: (c == 0 || c == 5) ? headLeft : headRight,
      );
    }

    // Item rows
    var r = headerRow + 1;
    for (final row in sheet.rows) {
      s.updateCell(at(0, r), xl.TextCellValue(row.displayName));
      s.updateCell(at(1, r), xl.IntCellValue(row.opening), cellStyle: number);
      s.updateCell(at(2, r), xl.IntCellValue(row.supply), cellStyle: number);
      s.updateCell(at(3, r), xl.IntCellValue(row.sale), cellStyle: number);
      s.updateCell(at(4, r), xl.IntCellValue(row.closing), cellStyle: number);
      if (row.remark.isNotEmpty) {
        s.updateCell(at(5, r), xl.TextCellValue(row.remark));
      }
      r++;
    }

    // Total row
    s.updateCell(at(0, r), xl.TextCellValue('Total'), cellStyle: totalLeft);
    s.updateCell(at(1, r), xl.IntCellValue(sheet.totalOpening), cellStyle: totalRight);
    s.updateCell(at(2, r), xl.IntCellValue(sheet.totalSupply), cellStyle: totalRight);
    s.updateCell(at(3, r), xl.IntCellValue(sheet.totalSale), cellStyle: totalRight);
    s.updateCell(at(4, r), xl.IntCellValue(sheet.totalClosing), cellStyle: totalRight);

    s.setColumnWidth(0, 44);
    for (var c = 1; c <= 4; c++) {
      s.setColumnWidth(c, 16);
    }
    s.setColumnWidth(5, 44);
  }
}

import 'package:supabase_flutter/supabase_flutter.dart' show SupabaseClient;

import '../core/app_exception.dart';

class ItemRepository {
  ItemRepository(this._db);
  final SupabaseClient _db;

  static const _select = '*, brands(id,name), categories(id,name), item_stock(quantity)';

  Future<List<Item>> fetchItems() async {
    final rows = await _db.from('items').select(_select).eq('is_active', true).order('name');
    return rows.map((r) => Item.fromJson(r)).toList();
  }

  Future<Item> fetchItem(String id) async {
    final row = await _db.from('items').select(_select).eq('id', id).maybeSingle();
    if (row == null) throw const AppException('Item not found.');
    return Item.fromJson(row);
  }

  Future<void> createItem(String userId, ItemInput input) async {
    await _db.from('items').insert({
      ...input.toJson(),
      'created_by': userId,
      'updated_by': userId,
    });
  }

  Future<void> updateItem(String id, String userId, ItemInput input) async {
    await _db
        .from('items')
        .update({
          ...input.toJson(),
          'updated_by': userId,
        })
        .eq('id', id);
  }

  /// "Delete" hides the item. History stays safe.
  Future<void> archiveItem(String id, String userId) async {
    await _db.from('items').update({'is_active': false, 'updated_by': userId}).eq('id', id);
  }

  Future<List<Lookup>> fetchBrands() => _fetchLookup('brands');
  Future<List<Lookup>> fetchCategories() => _fetchLookup('categories');
  Future<Lookup> createBrand(String name) => _createLookup('brands', name);
  Future<Lookup> createCategory(String name) => _createLookup('categories', name);

  Future<List<Lookup>> _fetchLookup(String table) async {
    final rows = await _db.from(table).select('id, name').order('name');
    return rows.map((r) => Lookup.fromJson(r)).toList();
  }

  Future<Lookup> _createLookup(String table, String name) async {
    final row = await _db.from(table).insert({'name': name.trim()}).select('id, name').single();
    return Lookup.fromJson(row);
  }
}

import 'package:intl/intl.dart';
import 'package:supabase_flutter/supabase_flutter.dart' show SupabaseClient;

import '../models/inventory_entry.dart';

class InventoryRepository {
  InventoryRepository(this._db);
  final SupabaseClient _db;

  static const _entrySelect =
      '*, app_users(display_name), '
      'inventory_entry_lines(id, quantity_change, item_id, items(name,size,variation))';

  static final _dateFormat = DateFormat('yyyy-MM-dd');
  static String _date(DateTime d) => _dateFormat.format(d);

  static String? _clean(String? s) {
    final t = s?.trim() ?? '';
    return t.isEmpty ? null : t;
  }

  /// quantities: itemId -> quantity.
  /// Supply and sale: positive numbers. Manual: signed (+ or -).
  Future<void> createEntry({
    required String userId,
    required EntryType type,
    required DateTime entryDate,
    required Map<String, int> quantities,
    String? referenceNo,
    String? note,
  }) async {
    await _db.rpc(
      'create_inventory_entry',
      params: {
        'p_user_id': userId,
        'p_entry_type': type.name,
        'p_reference_no': _clean(referenceNo),
        'p_entry_date': _date(entryDate),
        'p_note': _clean(note),
        'p_lines': quantities.entries.map((e) => {'item_id': e.key, 'quantity': e.value}).toList(),
      },
    );
  }

  Future<List<InventoryEntry>> fetchEntries({
    DateTime? from,
    DateTime? to,
    EntryType? type,
    int limit = 200,
  }) async {
    var query = _db.from('inventory_entries').select(_entrySelect);
    if (from != null) query = query.gte('entry_date', _date(from));
    if (to != null) query = query.lte('entry_date', _date(to));
    if (type != null) query = query.eq('entry_type', type.name);

    final rows = await query.order('entry_date', ascending: false).order('created_at', ascending: false).limit(limit);
    return rows.map((r) => InventoryEntry.fromJson(r)).toList();
  }

  Future<List<ItemHistoryRow>> fetchItemHistory(String itemId) async {
    final rows = await _db
        .from('inventory_entry_lines')
        .select(
          'quantity_change, '
          'inventory_entries(entry_type, reference_no, note, entry_date, created_at, '
          'app_users(display_name))',
        )
        .eq('item_id', itemId)
        .limit(500);

    final list = rows.map((r) => ItemHistoryRow.fromJson(r)).toList();
    list.sort((a, b) {
      final byDate = b.entryDate.compareTo(a.entryDate);
      return byDate != 0 ? byDate : b.createdAt.compareTo(a.createdAt);
    });
    return list;
  }

  /// Deletes a record and puts the stock back.
  /// Fails with a clear message if the stock would go below 0.
  Future<void> deleteEntry({required String userId, required String entryId}) async {
    await _db.rpc(
      'delete_inventory_entry',
      params: {'p_user_id': userId, 'p_entry_id': entryId},
    );
  }

  /// Raw report rows from the database (one row per item per date).
  Future<List<Map<String, dynamic>>> fetchStockReportRows(DateTime from, DateTime to) async {
    final res = await _db.rpc(
      'get_stock_report',
      params: {'p_from': _date(from), 'p_to': _date(to)},
    );
    final list = (res as List?) ?? const [];
    return list.map((e) => Map<String, dynamic>.from(e as Map)).toList();
  }
}

/// One item row of a daily stock sheet.
class ReportRow {
  const ReportRow({
    required this.itemId,
    required this.name,
    required this.subtitle,
    required this.opening,
    required this.supply,
    required this.sale,
    required this.closing,
    this.remark = '',
  });

  final String itemId;
  final String name;
  final String subtitle;
  final int opening;
  final int supply;
  final int sale;
  final int closing;
  final String remark;

  String get displayName => subtitle.isEmpty ? name : '$name ($subtitle)';

  /// True when something happened to this item on that day.
  bool get hasMovement => supply != 0 || sale != 0 || remark.isNotEmpty;

  factory ReportRow.fromJson(Map<String, dynamic> j) => ReportRow(
    itemId: j['item_id'] as String,
    name: (j['name'] as String?) ?? 'Unknown item',
    subtitle: joinSubtitle(
      (j['variation'] as String?) ?? '',
      (j['size'] as String?) ?? '',
    ),
    opening: (j['opening'] as num).toInt(),
    supply: (j['supply_qty'] as num).toInt(),
    sale: (j['sale_qty'] as num).toInt(),
    closing: (j['closing'] as num).toInt(),
    remark: (j['remark'] as String?) ?? '',
  );
}

/// One sheet = one date.
class ReportSheet {
  const ReportSheet({required this.date, required this.rows});

  final DateTime date;
  final List<ReportRow> rows;

  int get totalOpening => rows.fold(0, (sum, r) => sum + r.opening);
  int get totalSupply => rows.fold(0, (sum, r) => sum + r.supply);
  int get totalSale => rows.fold(0, (sum, r) => sum + r.sale);
  int get totalClosing => rows.fold(0, (sum, r) => sum + r.closing);
}

/// Builds one sheet per date. The database only sends rows that have stock or
/// movement. Active items that are missing are added here with zeros.
List<ReportSheet> buildReportSheets({
  required DateTime from,
  required DateTime to,
  required List<Map<String, dynamic>> serverRows,
  required List<Item> activeItems,
}) {
  final keyFormat = DateFormat('yyyy-MM-dd');

  final byDate = <String, Map<String, ReportRow>>{};
  for (final j in serverRows) {
    final row = ReportRow.fromJson(j);
    (byDate[j['date'] as String] ??= {})[row.itemId] = row;
  }

  final sheets = <ReportSheet>[];
  var day = DateTime(from.year, from.month, from.day);
  final last = DateTime(to.year, to.month, to.day);

  while (!day.isAfter(last)) {
    final rows = Map<String, ReportRow>.from(byDate[keyFormat.format(day)] ?? const {});

    for (final item in activeItems) {
      // Do not list items that were created after this date
      final created = DateTime(item.createdAt.year, item.createdAt.month, item.createdAt.day);
      if (created.isAfter(day)) continue;

      rows.putIfAbsent(
        item.id,
        () => ReportRow(
          itemId: item.id,
          name: item.name,
          subtitle: item.subtitle,
          opening: 0,
          supply: 0,
          sale: 0,
          closing: 0,
        ),
      );
    }

    final list = rows.values.toList()
      ..sort((a, b) {
        final byName = a.name.toLowerCase().compareTo(b.name.toLowerCase());
        return byName != 0 ? byName : a.subtitle.toLowerCase().compareTo(b.subtitle.toLowerCase());
      });

    sheets.add(ReportSheet(date: day, rows: list));
    day = DateTime(day.year, day.month, day.day + 1);
  }
  return sheets;
}

class AppUser {

    required this.id,
    required this.username,
    required this.displayName,

  final String id;

  factory AppUser.fromJson(Map<String, dynamic> j) => AppUser(

  Map<String, dynamic> toJson() => {
    'id': id,
    'username': username,
    'display_name': displayName,
  };
}
// models/item.dart
/// "strawberry · 500ml"
String joinSubtitle(String variation, String size) => [variation, size].where((s) => s.trim().isNotEmpty).join(' · ');

class Item {
  const Item({
    required this.id,
    required this.name,
    required this.shortDesc,
    required this.size,
    required this.variation,
    required this.isActive,
    required this.stock,
    required this.createdBy,
    required this.updatedBy,
    required this.createdAt,
    required this.updatedAt,
    this.itemSerial,
    this.brandId,
    this.brandName,
    this.categoryId,
    this.categoryName,
    this.imageUrl,
  });

  final String id;
  final String? itemSerial;
  final String name;
  final String shortDesc;
  final String size;
  final String variation;
  final bool isActive;
  final int stock;
  final String? brandId;
  final String? brandName;
  final String? categoryId;
  final String? categoryName;
  final String? imageUrl;
  final String createdBy;
  final String updatedBy;
  final DateTime createdAt;
  final DateTime updatedAt;

  String get subtitle => joinSubtitle(variation, size);

  factory Item.fromJson(Map<String, dynamic> j) {
    final brand = j['brands'] as Map<String, dynamic>?;
    final category = j['categories'] as Map<String, dynamic>?;

    // item_stock may come back as an object or a list with one object
    final stockRaw = j['item_stock'];
    final Map<String, dynamic>? stockMap = stockRaw is List
        ? (stockRaw.isEmpty ? null : stockRaw.first as Map<String, dynamic>)
        : stockRaw as Map<String, dynamic>?;

    return Item(
      id: j['id'] as String,
      itemSerial: j['item_serial'] as String?,
      name: j['name'] as String,
      shortDesc: (j['short_desc'] as String?) ?? '',
      size: (j['size'] as String?) ?? '',
      variation: (j['variation'] as String?) ?? '',
      isActive: (j['is_active'] as bool?) ?? true,
      stock: (stockMap?['quantity'] as int?) ?? 0,
      brandId: j['brand_id'] as String?,
      brandName: brand?['name'] as String?,
      categoryId: j['category_id'] as String?,
      categoryName: category?['name'] as String?,
      imageUrl: j['image_url'] as String?,
      createdBy: j['created_by'] as String,
      updatedBy: j['updated_by'] as String,
      createdAt: DateTime.parse(j['created_at'] as String).toLocal(),
      updatedAt: DateTime.parse(j['updated_at'] as String).toLocal(),
    );
  }
}

/// Values from the create/edit form.
class ItemInput {
  const ItemInput({
    required this.name,
    this.itemSerial,
    this.brandId,
    this.categoryId,
    this.imageUrl,
    this.shortDesc = '',
    this.size = '',
    this.variation = '',
  });

  final String? itemSerial;
  final String name;
  final String? brandId;
  final String? categoryId;
  final String? imageUrl;
  final String shortDesc;
  final String size;
  final String variation;

  Map<String, dynamic> toJson() => {
    // An empty serial is saved as "no serial" (null), never as an empty text.
    'item_serial': (itemSerial == null || itemSerial!.trim().isEmpty) ? null : itemSerial!.trim(),
    'name': name.trim(),
    'brand_id': brandId,
    'category_id': categoryId,
    'image_url': (imageUrl == null || imageUrl!.trim().isEmpty) ? null : imageUrl!.trim(),
    'short_desc': shortDesc.trim(),
    'size': size.trim(),
    'variation': variation.trim(),
  };
}

import 'item.dart' show joinSubtitle;

enum EntryType {
  supply,
  sale,
  manual
  ;

  String get label => switch (this) {
    EntryType.supply => 'Supply',
    EntryType.sale => 'Sale',
    EntryType.manual => 'Manual',
  };
}

class EntryLine {
  const EntryLine({
    required this.itemId,
    required this.itemName,
    required this.itemSubtitle,
    required this.quantityChange,
  });

  final String itemId;
  final String itemName;
  final String itemSubtitle;
  final int quantityChange;

  factory EntryLine.fromJson(Map<String, dynamic> j) {
    final item = (j['items'] as Map<String, dynamic>?) ?? const {};
    return EntryLine(
      itemId: j['item_id'] as String,
      itemName: (item['name'] as String?) ?? 'Unknown item',
      itemSubtitle: joinSubtitle(
        (item['variation'] as String?) ?? '',
        (item['size'] as String?) ?? '',
      ),
      quantityChange: j['quantity_change'] as int,
    );
  }
}

class InventoryEntry {
  const InventoryEntry({
    required this.id,
    required this.type,
    required this.entryDate,
    required this.createdByName,
    required this.createdAt,
    required this.lines,
    this.referenceNo,
    this.note,
  });

  final String id;
  final EntryType type;
  final String? referenceNo;
  final String? note;
  final DateTime entryDate;
  final String createdByName;
  final DateTime createdAt;
  final List<EntryLine> lines;

  int get totalUnits => lines.fold(0, (sum, l) => sum + l.quantityChange.abs());

  factory InventoryEntry.fromJson(Map<String, dynamic> j) {
    final user = j['app_users'] as Map<String, dynamic>?;
    final rawLines = (j['inventory_entry_lines'] as List?) ?? const [];
    return InventoryEntry(
      id: j['id'] as String,
      type: EntryType.values.byName(j['entry_type'] as String),
      referenceNo: j['reference_no'] as String?,
      note: j['note'] as String?,
      entryDate: DateTime.parse(j['entry_date'] as String),
      createdByName: (user?['display_name'] as String?) ?? 'Unknown',
      createdAt: DateTime.parse(j['created_at'] as String).toLocal(),
      lines: rawLines.map((l) => EntryLine.fromJson(l as Map<String, dynamic>)).toList(),
    );
  }
}

/// One stock change for one item, used in Item Detail history.
class ItemHistoryRow {
  const ItemHistoryRow({
    required this.type,
    required this.quantityChange,
    required this.entryDate,
    required this.createdAt,
    required this.createdByName,
    this.referenceNo,
    this.note,
  });

  final EntryType type;
  final int quantityChange;
  final DateTime entryDate;
  final DateTime createdAt;
  final String createdByName;
  final String? referenceNo;
  final String? note;

  factory ItemHistoryRow.fromJson(Map<String, dynamic> j) {
    final e = j['inventory_entries'] as Map<String, dynamic>;
    final user = e['app_users'] as Map<String, dynamic>?;
    return ItemHistoryRow(
      type: EntryType.values.byName(e['entry_type'] as String),
      quantityChange: j['quantity_change'] as int,
      entryDate: DateTime.parse(e['entry_date'] as String),
      createdAt: DateTime.parse(e['created_at'] as String).toLocal(),
      createdByName: (user?['display_name'] as String?) ?? 'Unknown',
      referenceNo: e['reference_no'] as String?,
      note: e['note'] as String?,
    );
  }
}
